/** * Google Apps Script EduAI-Analytics Live * Output: JSON biasa dan JSONP jika callback wujud. */ const SHEET_NAME = ""; // Biarkan kosong untuk guna sheet aktif. // Jika perlu, letak nama tab contoh: "Form Responses 1" function doGet(e) { try { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!spreadsheet) { return jsonOutput({ success: false, error: "Spreadsheet tidak dijumpai." }, e); } const sheet = SHEET_NAME ? spreadsheet.getSheetByName(SHEET_NAME) : spreadsheet.getActiveSheet(); if (!sheet) { return jsonOutput({ success: false, error: "Sheet tidak dijumpai." }, e); } const values = sheet.getDataRange().getValues(); if (!values || values.length === 0) { return jsonOutput({ success: false, error: "Helaian Google Sheet kosong." }, e); } const headers = values[0].map(function(header) { return String(header || "").trim(); }); const rows = []; for (let i = 1; i < values.length; i++) { const row = {}; let hasData = false; for (let j = 0; j < headers.length; j++) { const header = headers[j]; if (!header) continue; let cellValue = values[i][j]; if (cellValue instanceof Date) { cellValue = Utilities.formatDate( cellValue, Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ss" ); } if (cellValue === null || cellValue === undefined) { cellValue = ""; } row[header] = cellValue; if (String(cellValue).trim() !== "") { hasData = true; } } if (hasData) { rows.push(row); } } return jsonOutput({ success: true, total: rows.length, sheetName: sheet.getName(), headers: headers, data: rows, updatedAt: Utilities.formatDate( new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ss" ) }, e); } catch (err) { return jsonOutput({ success: false, error: err.toString() }, e); } } function jsonOutput(obj, e) { const json = JSON.stringify(obj); let callback = null; if ( e && e.parameter && typeof e.parameter.callback === "string" && e.parameter.callback.trim() !== "" ) { callback = e.parameter.callback.trim(); } const validCallback = callback && /^[a-zA-Z_$][0-9a-zA-Z_$]*(\.[a-zA-Z_$][0-9a-zA-Z_$]*)*$/.test(callback); if (validCallback) { return ContentService .createTextOutput(callback + "(" + json + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); }
